    (function () {
      function initProofQuote() {
        var root = document.querySelector("[data-proof-quote]");
        if (!root || root.dataset.proofQuoteReady === "true") return;
        root.dataset.proofQuoteReady = "true";

        var slides = Array.prototype.slice.call(root.querySelectorAll("[data-proof-quote-slide]"));
        var dots = Array.prototype.slice.call(root.querySelectorAll("[data-proof-quote-dot]"));
        var prev = root.querySelector("[data-proof-quote-prev]");
        var next = root.querySelector("[data-proof-quote-next]");
        var activeIndex = 0;
        var timer = null;

        if (!slides.length) return;

        function setActive(index) {
          activeIndex = (index + slides.length) % slides.length;

          slides.forEach(function (slideNode, slideIndex) {
            slideNode.classList.toggle("is-active", slideIndex === activeIndex);
          });

          dots.forEach(function (dotNode, dotIndex) {
            dotNode.classList.toggle("is-active", dotIndex === activeIndex);
            dotNode.setAttribute("aria-pressed", dotIndex === activeIndex ? "true" : "false");
          });
        }

        function stopTimer() {
          if (timer) {
            window.clearInterval(timer);
            timer = null;
          }
        }

        function startTimer() {
          stopTimer();
          timer = window.setInterval(function () {
            setActive(activeIndex + 1);
          }, 4600);
        }

        if (prev) {
          prev.addEventListener("click", function () {
            setActive(activeIndex - 1);
            startTimer();
          });
        }

        if (next) {
          next.addEventListener("click", function () {
            setActive(activeIndex + 1);
            startTimer();
          });
        }

        dots.forEach(function (dotNode, dotIndex) {
          dotNode.addEventListener("click", function () {
            setActive(dotIndex);
            startTimer();
          });
        });

        root.addEventListener("mouseenter", stopTimer);
        root.addEventListener("mouseleave", startTimer);

        setActive(0);
        startTimer();
      }

      if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", initProofQuote, { once: true });
      } else {
        initProofQuote();
      }
    })();

          buttonNode.setAttribute("aria-pressed", isActive ? "true" : "false");
        });
        detail.innerHTML = teamPanelMarkup(pod);
      }

      function wireTeamTabs() {
        const buttons = Array.from(document.querySelectorAll(".weflair-demand-team__tab[data-pod-index]"));
        if (!buttons.length) return;
        buttons.forEach(function (buttonNode) {
          buttonNode.addEventListener("click", function () {
            const index = Number(buttonNode.getAttribute("data-pod-index"));
            if (Number.isNaN(index)) return;
            renderTeamPod(index);
          });
        });
      }

      function upgradePage() {
        ensureCalEmbed();

        const widgetWrap = document.querySelector(".weflair-hero-widget-wrap");
        if (widgetWrap) {
          widgetWrap.innerHTML = renderAuditWidget();
          wireAuditWidget();
        }

        applyCalAttrs(document.querySelector(".nav-bar__btn a"), "nav-talk");

        const servicesCta = document.querySelector(".weflair-services-cta");
        if (servicesCta && !servicesCta.classList.contains('weflair-questionnaire-hook')) {
          servicesCta.innerHTML = '<div class="weflair-services-cta-text"><h3 class="h5">Not sure which service you need?</h3><p>Answer three quick questions and we&rsquo;ll recommend the right starting point.</p></div>';
        }

        bindCalTriggers(document);
        document.querySelectorAll("#problems .growing-tile__end .growing-tile__text, #problems .growing-tile__end .p-s").forEach((node) => {
          node.style.visibility = "visible";
          node.style.opacity = "1";
          node.style.transform = "none";
        });
        wireTeamTabs();
      }

      if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", upgradePage, {
          once: true
        });
      } else {
        upgradePage();
      }
    })();
