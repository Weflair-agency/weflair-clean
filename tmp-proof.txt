          function metricHtml(metric) {

          }

          function railHtml(item, index) {
            var label = item.alt.replace(" logo", "");

            );

          function dotHtml(index) {

              (index === activeIndex ? " is-active" : "") +
              '" data-proof-dot="' +
              index +

              (index + 1) +
              '"></button>'
            );
          }

          function nextIndexes(start) {
            var indexes = [];
            for (var offset = 1; offset <= 4; offset += 1) {
              indexes.push((start + offset) % proofCases.length);
            }
            return indexes;
          }

          function render() {
            var item = proofCases[activeIndex];

            // set card type
            featureEl.setAttribute("data-type", item.type);
            featureEl.setAttribute("data-layout", item.layout || "default");

            if (item.type === "heydigital") {
              // populate heydigital elements
              if (logoTarget) logoTarget.innerHTML = logoHtml(item);
              if (headlineTarget)
                headlineTarget.textContent = item.headline || "";
              if (descTarget) descTarget.textContent = item.desc;
              if (metricsTarget)
                metricsTarget.innerHTML = item.metrics.map(metricHtml).join("");
              if (linkTarget) linkTarget.setAttribute("href", item.link);

              // photo or ad-showcase
              if (item.visual === "photo" && photoTarget) {
                if (item.photo) {
                  photoTarget.style.backgroundImage =
                    'url("' + item.photo + '")';
                  photoTarget.classList.remove("no-photo");
                } else {
                  photoTarget.style.backgroundImage = "none";
                  photoTarget.classList.add("no-photo");
                }
                photoTarget.style.backgroundPosition =
                  item.photoPosition || "center center";
                photoTarget.parentElement.style.display = "";
                if (adGridTarget) adGridTarget.style.display = "none";
              } else if (
                item.visual === "ad-showcase" &&
                adGridTarget &&
                item.adFrames
              ) {
                adGridTarget.innerHTML = item.adFrames
                  .map(function (frame) {
                    return adFrameHtml(frame, item);
                  })
                  .join("");
                adGridTarget.style.display = "";
                if (photoTarget)
                  photoTarget.parentElement.style.display = "none";
              }
            } else if (item.type === "netpeak") {
              // populate netpeak elements
              if (npLogoTarget) npLogoTarget.innerHTML = logoHtml(item);
              if (npPhotoTarget) {
                if (item.photo) {
                  npPhotoTarget.style.backgroundImage =
                    'url("' + item.photo + '")';
                  npPhotoTarget.classList.remove("no-photo");
                } else {
                  npPhotoTarget.style.backgroundImage = "none";
                  npPhotoTarget.classList.add("no-photo");
                }
              }
              if (npDescTarget) npDescTarget.textContent = item.desc;
              if (npStatsTarget)
                npStatsTarget.innerHTML = item.metrics.map(npStatHtml).join("");
              if (npCtaTarget) npCtaTarget.setAttribute("href", item.link);

              // services + duration meta
              if (npMetaTarget) {
                var metaHtml = "";
                if (item.services) {
                  metaHtml +=
                    '<div><div class="weflair-proof-feature__np-meta-label">Services delivered</div><div class="weflair-proof-feature__np-meta-value">' +
                    item.services.join(", ") +
                    "</div></div>";
                }
                if (item.duration) {
                  metaHtml +=
                    '<div><div class="weflair-proof-feature__np-meta-label">Partnership duration</div><div class="weflair-proof-feature__np-meta-duration">' +
                    item.duration +
                    "</div></div>";
                }
                npMetaTarget.innerHTML = metaHtml;
              }
            }

            // rails
            if (railsTarget) {
              railsTarget.innerHTML = nextIndexes(activeIndex)
                .map(function (index) {
                  return railHtml(proofCases[index], index);
                })
                .join("");
            }

            // dots
            if (dotsTarget) {
              dotsTarget.innerHTML = proofCases
                .map(function (_, index) {
                  return dotHtml(index);
                })
                .join("");
            }

            // bind rails
            Array.prototype.forEach.call(
              root.querySelectorAll("[data-proof-select]"),
              function (button) {
                button.addEventListener("click", function () {
                  activeIndex = Number(
                    button.getAttribute("data-proof-select"),
                  );
                  render();
                });
              },
            );

            // bind dots
            Array.prototype.forEach.call(
              root.querySelectorAll("[data-proof-dot]"),
              function (button) {
                button.addEventListener("click", function () {
                  activeIndex = Number(button.getAttribute("data-proof-dot"));
                  render();
                });

          }

            activeIndex =
              (activeIndex - 1 + proofCases.length) % proofCases.length;
            render();
          });

          nextButton.addEventListener("click", function () {
            activeIndex = (activeIndex + 1) % proofCases.length;
            render();
